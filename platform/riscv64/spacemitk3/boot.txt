# 1. 设置 ip 地址
setenv serverip 192.168.137.1;
setenv ipaddr 192.168.137.200;
setenv netmask 255.255.255.0;

# 2. 从 TFTP 下载三个文件（注意：地址均已升级为 64 位）
tftpboot 0x102200000 Image
tftpboot 0x105000000 minimal.dtb
tftpboot 0x106000000 initrd.cpio

# 3. 设置内核启动参数
setenv bootargs "root=/dev/sda3 rw console=ttyS0,115200 earlycon"

# 4. 点火启动 (${filesize} 会自动获取刚下载完的 initrd 大小)
booti 0x102200000 0x106000000:${filesize} 0x105000000
